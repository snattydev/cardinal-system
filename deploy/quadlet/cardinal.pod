[Unit]
Description=Cardinal System pod - backend, frontend, MCP Context Hub, and Caddy reverse proxy
After=network-online.target

[Pod]
PodName=cardinal
# Publish Caddy's port 80 to host 127.0.0.1:8080 (local dev)
# For production, add :443 to expose HTTPS:
# PublishPort=127.0.0.1:443:443
PublishPort=127.0.0.1:8080:80
# MCP Context Hub: loopback only, never exposed beyond the host
PublishPort=127.0.0.1:8789:8789
# Map container UID/GID 1000 (USER bun) to the invoking host user so the
# read-only vault and writable agent-state bind mounts resolve correctly for
# non-root processes, while root inside the pod namespace still works for
# nginx and Caddy.
UserNS=keep-id:uid=1000,gid=1000

[Install]
WantedBy=default.target
